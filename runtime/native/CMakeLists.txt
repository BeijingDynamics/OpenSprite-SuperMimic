# Build separate replay, receive-only and protected IPC actors against CPU ORT.
set(SPRITE_ORT_INCLUDE "" CACHE PATH "ONNX Runtime C++ headers")
set(SPRITE_ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime shared library")
if(SPRITE_ORT_INCLUDE AND SPRITE_ORT_LIBRARY)
  add_executable(sprite_policy_replay src/sprite_policy_replay.cpp)
  target_include_directories(sprite_policy_replay PRIVATE "${SPRITE_ORT_INCLUDE}")
  target_link_libraries(sprite_policy_replay PRIVATE "${SPRITE_ORT_LIBRARY}")
  target_compile_options(sprite_policy_replay PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  add_executable(sprite_supermimic_replay src/sprite_supermimic_replay.cpp)
  target_include_directories(sprite_supermimic_replay PRIVATE "${SPRITE_ORT_INCLUDE}")
  target_link_libraries(sprite_supermimic_replay PRIVATE "${SPRITE_ORT_LIBRARY}" pthread)
  target_compile_options(sprite_supermimic_replay PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  add_executable(sprite_supermimic_ipc_readonly src/sprite_supermimic_replay.cpp)
  target_compile_definitions(sprite_supermimic_ipc_readonly PRIVATE SPRITE_SUPERMIMIC_IPC_READONLY)
  target_include_directories(sprite_supermimic_ipc_readonly PRIVATE "${SPRITE_ORT_INCLUDE}")
  target_link_libraries(sprite_supermimic_ipc_readonly PRIVATE "${SPRITE_ORT_LIBRARY}" pthread)
  target_compile_options(sprite_supermimic_ipc_readonly PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
  add_executable(sprite_supermimic_protected_ipc src/sprite_supermimic_replay.cpp)
  target_compile_definitions(sprite_supermimic_protected_ipc PRIVATE SPRITE_SUPERMIMIC_PROTECTED_IPC)
  target_include_directories(sprite_supermimic_protected_ipc PRIVATE "${SPRITE_ORT_INCLUDE}")
  target_link_libraries(sprite_supermimic_protected_ipc PRIVATE "${SPRITE_ORT_LIBRARY}" pthread)
  target_compile_options(sprite_supermimic_protected_ipc PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
endif()
